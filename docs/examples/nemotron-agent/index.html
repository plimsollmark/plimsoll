What happened in this session

I gave Nemotron 3 Super (120B) one task and one tool: run_code, which runs code in the sandbox and hands back the output.

The assignment

Write a Python function iso_duration_seconds(s) that converts an ISO 8601 duration string to a number of seconds (a float). It must handle: "PT0S" -> 0, "PT1H30M" -> 5400, "P1DT2H" -> 93600, "P1W" -> 604800, "PT0.5S" -> 0.5, "P2DT3H4M5.25S" -> 183845.25, and it must raise ValueError for "", "P", "PT", "1H" and "P1H". Use 1 day = 86400 seconds and 1 week = 7 days; years and months are not supported and must raise ValueError. Write the function and tests for every case above, run the tests, and fix the code until every test passes. Then report the final test output.

import re

def iso_duration_seconds(s):
    if not isinstance(s, str):
        raise ValueError("Input must be string")
    if s == "" or s == "P" or s == "PT":
        raise ValueError("Invalid duration")
    # Must start with P
    if not s.startswith('P'):
        raise ValueError("Duration must start with 'P'")
    # Split date and time parts
    if 'T' in s:
        date_part, time_part = s[1:].split('T', 1)
    else:
        date_part, time_part = s[1:], ''
    # Validate no unsupported units Y or M anywhere
    if re.search(r'[YM]', s):
        raise ValueError("Years and months not supported")
    total = 0.0
    # Parse date part: only D allowed
    if date_part:
        # Expect pattern like (\d+(\.\d+)?)D
        m = re.fullmatch(r'(\d+(?:\.\d+)?)D', date_part)
        if not m:
            raise ValueError("Invalid date part")
        days = float(m.group(1))
        total += days * 86400
    # Parse time part: H, M, S allowed, order not enforced but we parse sequentially
    if time_part:
        # Remove possible empty
        # Use regex to find all number+unit
        tokens = re.findall(r'(\d+(?:\.\d+)?)([HMS])', time_part)
        if not tokens:
            raise ValueError("Invalid time part")
        # Ensure we consumed whole string
        reconstructed = ''.join(num + unit for num, unit in tokens)
        if reconstructed != time_part:
            raise ValueError("Invalid time part format")
        for num, unit in tokens:
            val = float(num)
            if unit == 'H':
                total += val * 3600
            elif unit == 'M':
                total += val * 60
            elif unit == 'S':
                total += val
            else:
                raise ValueError("Unsupported unit")
    # Ensure at least one component contributed
    if total == 0 and s not in ("PT0S",):  # allow zero explicitly
        # but also PT0S should be allowed
        # check if any number present
        if not re.search(r'\d', s):
            raise ValueError("No duration specified")
    return total

import re

def iso_duration_seconds(s):
    if not isinstance(s, str):
        raise ValueError("Input must be string")
    if s == "" or s == "P" or s == "PT":
        raise ValueError("Invalid duration")
    if not s.startswith('P'):
        raise ValueError("Duration must start with 'P'")
    rest = s[1:]
    # Split into date and time parts
    if 'T' in rest:
        date_part, time_part = rest.split('T', 1)
    else:
        date_part, time_part = rest, ''
    # Years and months not allowed anywhere
    if re.search(r'[YM]', s):
        # But allow M as minute only if after T? Simpler: reject any M or Y.
        # However we need to allow M for minutes. So we check for Y only.
        if 'Y' in s:
            raise ValueError("Years and months not supported")
        # For M, we need to ensure it's not a month. Since we don't support months,
        # any M appearing before T is month. So check date_part for M.
        if 'M' in date_part:
            raise ValueError("Years and months not supported")
    total = 0.0
    # Parse date part: D and/or W allowed
    if date_part:
        # Find all number+unit pairs
        matches = re.findall(r'(\d+(?:\.\d+)?)([DW])', date_part)
        if not matches:
            raise ValueError("Invalid date part")
        # Ensure we consumed the whole date_part
        reconstructed = ''.join(num + unit for num, unit in matches)
        if reconstructed != date_part:
            raise ValueError("Invalid date part format")
        for num, unit in matches:
            val = float(num)
            if unit == 'D':
                total += val * 86400
            elif unit == 'W':
                total += val * 7 * 86400
            else:
                raise ValueError("Unsupported date unit")
    # Parse time part: H, M, S allowed
    if time_part:
        matches = re.findall(r'(\d+(?:\.\d+)?)([HMS])', time_part)
        if not matches:
            raise ValueError("Invalid time part")
        reconstructed = ''.join(num + unit for num, unit in matches)
        if reconstructed != time_part:
            raise ValueError("Invalid time part format")
        for num, unit in matches:
            val = float(num)
            if unit == 'H':
                total += val * 3600
            elif unit == 'M':
                total += val * 60
            elif unit == 'S':
                total += val
            else:
                raise ValueError("Unsupported time unit")
    # Ensure at least one component contributed (except PT0S already handled)
    if total == 0:
        # Check if there was any number at all
        if not re.search(r'\d', s):
            raise ValueError("No duration specified")
    return total

What this proves, and what it doesn't

Proves
  • Exactly what code was sent, what came back, and which sandbox and isolation tier the daemon reported.
  • The order of the runs, and that none was changed or dropped afterward.
  • That a claim and its evidence match, or don't, for anyone who checks.
Doesn't prove
  • That the code is correct. A grocery receipt proves what you bought, not that dinner was good.
  • That the hardware is honest: the isolation tier is the daemon's configuration and startup checks, not a hardware attestation.
  • Anything about runs that never went through plimsoll.
